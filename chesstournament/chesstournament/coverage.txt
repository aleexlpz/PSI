from django.db import models
class Usuario(models.Model):
	nombreUsuario = models.CharField(max_length=100)
	def __str__(self):
       		return self.nombreUsuario
class Canal(models.Model):
	nombreCanal = models.CharField(max_length=100)
	def __str__(self):
       		return self.nombreCanal
class Suscripcion(models.Model):
	canal = models.ForeignKey(Canal, on_delete=models.CASCADE)
	usuario = models.ForeignKey(Usuario, on_delete=models.CASCADE)
	fechaDeSuscripcion = models.DateField(auto_now=True)
	def __str__(self):
       		return self

from django.shortcuts import render
from .models import Usuario, Canal, Suscripcion
import datetime

def canal(request, pk):
   """
   Función vista para la página inicio del sitio.
   """
   error = None
   nombreCanal = None
   suscripciones = None
   try:
       canal = Canal.objects.get(pk=pk)
       nombreCanal = canal.nombreCanal
       suscripciones = Suscripcion.objects.filter(canal=canal)
   except Canal.DoesNotExist:
       error = "No existe un canal con este id"
   context = {
       'nombreCanal': nombreCanal,
       'suscripciones': suscripciones,
       'error': error,
   }
   return render(request, 'canal.html', context)

-modificamos las vistas de la api:

from application.models import Canal, Usuario, Suscripcion
from .serializers import CanalSerializer, UsuarioSerializer, SuscripcionSerializer
from rest_framework import viewsets
class canalViewSet(viewsets.ModelViewSet):
   queryset = Canal.objects.all()
   serializer_class = CanalSerializer
class usuarioViewSet(viewsets.ModelViewSet):
   queryset = Usuario.objects.all()
   serializer_class = UsuarioSerializer
class suscripcionViewSet(viewsets.ModelViewSet):
   queryset = Suscripcion.objects.all()
   serializer_class = SuscripcionSerializer
